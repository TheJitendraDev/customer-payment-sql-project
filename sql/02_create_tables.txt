CREATE TABLE customer (
    customer_id BIGINT PRIMARY KEY,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100),
    address_id BIGINT
);

CREATE TABLE payment (
    payment_id BIGINT PRIMARY KEY,
    customer_id BIGINT,
    amount NUMERIC(10,2),
    mode VARCHAR(50),
    payment_date DATE
);
